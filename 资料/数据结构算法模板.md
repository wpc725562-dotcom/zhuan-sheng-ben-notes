## 1. 线性表（Linear List）

### 1.1 顺序表定义（数组实现）

```c
#define MAXSIZE 100          // 顺序表最大长度

typedef struct {
    int data[MAXSIZE];       // 用数组存储数据元素
    int length;              // 顺序表当前长度
} SeqList;
```

- **时间复杂度**：O(1)（定义/初始化）
- **稳定性**：—

---

### 1.2 顺序表插入操作

```c
/**
 * 在顺序表 L 的第 i 个位置（1 ≤ i ≤ L.length+1）插入元素 e
 * 成功返回 1，失败返回 0
 */
int SeqListInsert(SeqList *L, int i, int e) {
    int j;

    // 1. 检查插入位置是否合法
    if (i < 1 || i > L->length + 1) {
        return 0;                    // 位置不合法
    }
    // 2. 检查表是否已满
    if (L->length >= MAXSIZE) {
        return 0;                    // 表已满，无法插入
    }
    // 3. 将第 i 个位置及之后的元素后移
    for (j = L->length; j >= i; j--) {
        L->data[j] = L->data[j - 1];
    }
    // 4. 插入新元素
    L->data[i - 1] = e;
    // 5. 表长加 1
    L->length++;

    return 1;
}
```

- **时间复杂度**：O(n)（平均移动 n/2 个元素）
- **稳定性**：—

---

### 1.3 顺序表删除操作

```c
/**
 * 删除顺序表 L 中第 i 个位置（1 ≤ i ≤ L.length）的元素
 * 被删除的元素通过 *e 返回
 * 成功返回 1，失败返回 0
 */
int SeqListDelete(SeqList *L, int i, int *e) {
    int j;

    // 1. 检查删除位置是否合法
    if (i < 1 || i > L->length) {
        return 0;
    }
    // 2. 取出被删除元素
    *e = L->data[i - 1];
    // 3. 将第 i 个位置之后的元素前移
    for (j = i; j < L->length; j++) {
        L->data[j - 1] = L->data[j];
    }
    // 4. 表长减 1
    L->length--;

    return 1;
}
```

- **时间复杂度**：O(n)（平均移动 (n-1)/2 个元素）
- **稳定性**：—

---

### 1.4 顺序表查找操作

```c
/**
 * 在顺序表 L 中查找值为 e 的元素
 * 返回第一个匹配元素的位置（从 1 开始），未找到返回 0
 */
int SeqListSearch(SeqList *L, int e) {
    int i;

    for (i = 0; i < L->length; i++) {
        if (L->data[i] == e) {
            return i + 1;            // 返回位序（从 1 开始）
        }
    }
    return 0;                        // 未找到
}
```

- **时间复杂度**：O(n)
- **稳定性**：—

### 1.6 单链表创建（头插法 / 尾插法）

#### 头插法

```c
/**
 * 头插法创建单链表：读入 n 个元素，每次将新节点插入到头节点之后
 * 最终顺序与输入顺序相反
 */
LinkList CreateList_Head(int n) {
    LinkList L;                      // 头指针
    LNode *s;                        // 新节点指针
    int i;

    // 1. 创建头节点
    L = (LinkList)malloc(sizeof(LNode));
    L->next = NULL;                  // 初始化为空链表

    // 2. 循环读入 n 个元素
    for (i = 0; i < n; i++) {
        // 3. 创建新节点
        s = (LNode *)malloc(sizeof(LNode));
        scanf("%d", &s->data);       // 读入数据

        // 4. 将新节点插入到头节点之后（头插）
        s->next = L->next;
        L->next = s;
    }

    return L;
}
```

- **时间复杂度**：O(n)
- **稳定性**：—

#### 尾插法

```c
/**
 * 尾插法创建单链表：读入 n 个元素，每次将新节点插入到链表尾部
 * 最终顺序与输入顺序相同
 */
LinkList CreateList_Tail(int n) {
    LinkList L;                      // 头指针
    LNode *s, *r;                    // s 为新节点，r 为尾指针
    int i;

    // 1. 创建头节点
    L = (LinkList)malloc(sizeof(LNode));
    r = L;                           // 初始时尾指针指向头节点

    // 2. 循环读入 n 个元素
    for (i = 0; i < n; i++) {
        // 3. 创建新节点
        s = (LNode *)malloc(sizeof(LNode));
        scanf("%d", &s->data);       // 读入数据

        // 4. 将新节点插入到尾部
        r->next = s;
        r = s;                       // 尾指针后移
    }
    r->next = NULL;                  // 最后一个节点的 next 置空

    return L;
}
```

- **时间复杂度**：O(n)
- **稳定性**：—

### 1.7 单链表插入节点

```c
/**
 * 在单链表 L 的第 i 个位置（1 ≤ i ≤ 表长+1）插入值为 e 的节点
 * 成功返回 1，失败返回 0
 */
int ListInsert(LinkList L, int i, int e) {
    LNode *p = L;                    // p 指向头节点
    LNode *s;
    int j = 0;                       // 计数器，记录当前 p 指向的是第几个节点

    // 1. 查找第 i-1 个节点（插入位置的前驱）
    while (p != NULL && j < i - 1) {
        p = p->next;
        j++;
    }
    // 2. 检查位置是否合法
    if (p == NULL || j > i - 1) {
        return 0;                    // 位置不合法
    }
    // 3. 创建新节点
    s = (LNode *)malloc(sizeof(LNode));
    s->data = e;
    // 4. 插入节点：新节点指向前驱的 next，前驱的 next 指向新节点
    s->next = p->next;
    p->next = s;

    return 1;
}
```

- **时间复杂度**：O(n)（查找第 i-1 个节点为 O(n)，插入本身为 O(1)）
- **稳定性**：—

---

### 1.8 单链表删除节点

```c
/**
 * 删除单链表 L 中第 i 个位置（1 ≤ i ≤ 表长）的节点
 * 被删除节点的数据通过 *e 返回
 * 成功返回 1，失败返回 0
 */
int ListDelete(LinkList L, int i, int *e) {
    LNode *p = L;                    // p 指向头节点
    LNode *q;
    int j = 0;

    // 1. 查找第 i-1 个节点（被删节点的前驱）
    while (p->next != NULL && j < i - 1) {
        p = p->next;
        j++;
    }
    // 2. 检查位置是否合法
    if (p->next == NULL || j > i - 1) {
        return 0;                    // 位置不合法
    }
    // 3. 令 q 指向被删节点，取出数据
    q = p->next;
    *e = q->data;
    // 4. 删除节点：前驱的 next 指向被删节点的 next
    p->next = q->next;
    // 5. 释放被删节点内存
    free(q);

    return 1;
}
```

- **时间复杂度**：O(n)（查找前驱为 O(n)，删除本身为 O(1)）
- **稳定性**：—

## 2. 栈（Stack）

### 2.1 顺序栈定义

```c
#define MAXSIZE 100          // 栈的最大容量

typedef struct {
    int data[MAXSIZE];       // 用数组存储栈元素
    int top;                 // 栈顶指针（下标），-1 表示空栈
} SqStack;
```

- **时间复杂度**：O(1)（定义）
- **稳定性**：—

---

### 2.2 入栈操作（Push）

```c
/**
 * 将元素 e 压入栈 S
 * 成功返回 1，失败返回 0（栈满）
 */
int Push(SqStack *S, int e) {
    // 检查栈是否已满
    if (S->top == MAXSIZE - 1) {
        return 0;                    // 栈满
    }
    // 栈顶指针先加 1，再赋值
    S->top++;
    S->data[S->top] = e;

    return 1;
}
```

- **时间复杂度**：O(1)
- **稳定性**：—

---

### 2.3 出栈操作（Pop）

```c
/**
 * 弹出栈 S 的栈顶元素，通过 *e 返回
 * 成功返回 1，失败返回 0（空栈）
 */
int Pop(SqStack *S, int *e) {
    // 检查栈是否为空
    if (S->top == -1) {
        return 0;                    // 空栈
    }
    // 先取出栈顶元素，栈顶指针再减 1
    *e = S->data[S->top];
    S->top--;

    return 1;
}
```

- **时间复杂度**：O(1)
- **稳定性**：—

---

### 2.4 取栈顶元素（Top）

```c
/**
 * 读取栈 S 的栈顶元素，通过 *e 返回（不改变栈）
 * 成功返回 1，失败返回 0（空栈）
 */
int GetTop(SqStack *S, int *e) {
    if (S->top == -1) {
        return 0;                    // 空栈
    }
    *e = S->data[S->top];            // 直接读取栈顶元素

    return 1;
}
```

- **时间复杂度**：O(1)
- **稳定性**：—

## 3. 队列（Queue）

### 3.1 顺序队列定义（循环队列）

```c
#define MAXSIZE 100          // 队列最大容量

typedef struct {
    int data[MAXSIZE];       // 用数组存储队列元素
    int front;               // 队头指针（下标）
    int rear;                // 队尾指针（下标），指向队尾元素的下一个位置
} SqQueue;
```

> **循环队列核心**：`(rear + 1) % MAXSIZE == front` 表示队满，`rear == front` 表示队空。  
> 循环队列中最多存储 `MAXSIZE - 1` 个元素（牺牲一个单元区分队空/队满）。

- **时间复杂度**：O(1)（定义）
- **稳定性**：—

---

### 3.2 入队操作（Enqueue）

```c
/**
 * 将元素 e 插入循环队列 Q 的队尾
 * 成功返回 1，失败返回 0（队满）
 */
int EnQueue(SqQueue *Q, int e) {
    // 检查队列是否已满（牺牲一个单元）
    if ((Q->rear + 1) % MAXSIZE == Q->front) {
        return 0;                    // 队满
    }
    // 将元素放入队尾位置，然后 rear 指针后移（循环）
    Q->data[Q->rear] = e;
    Q->rear = (Q->rear + 1) % MAXSIZE;

    return 1;
}
```

- **时间复杂度**：O(1)
- **稳定性**：—

---

### 3.3 出队操作（Dequeue）

```c
/**
 * 从循环队列 Q 中取出队头元素，通过 *e 返回
 * 成功返回 1，失败返回 0（空队）
 */
int DeQueue(SqQueue *Q, int *e) {
    // 检查队列是否为空
    if (Q->front == Q->rear) {
        return 0;                    // 空队
    }
    // 取出队头元素，然后 front 指针后移（循环）
    *e = Q->data[Q->front];
    Q->front = (Q->front + 1) % MAXSIZE;

    return 1;
}
```

- **时间复杂度**：O(1)
- **稳定性**：—

### 4.5 求二叉树深度

```c
/**
 * 计算二叉树的深度（高度）
 * 空树深度为 0，非空树深度 = max(左子树深度, 右子树深度) + 1
 */
int TreeDepth(BiTree T) {
    int leftDepth, rightDepth;

    if (T == NULL) {
        return 0;                    // 空树深度为 0
    }

    leftDepth = TreeDepth(T->lchild);   // 求左子树深度
    rightDepth = TreeDepth(T->rchild);  // 求右子树深度

    // 返回左右子树中较大的深度 + 1
    return (leftDepth > rightDepth ? leftDepth : rightDepth) + 1;
}
```

- **时间复杂度**：O(n)（每个节点访问一次）
- **稳定性**：—

---

### 4.6 求叶子节点数

```c
/**
 * 统计二叉树中叶子节点（度为 0 的节点）的个数
 */
int LeafCount(BiTree T) {
    if (T == NULL) {
        return 0;                    // 空树，叶子节点数为 0
    }
    // 左右子树都为空，则是叶子节点
    if (T->lchild == NULL && T->rchild == NULL) {
        return 1;                    // 叶子节点
    }
    // 非叶子节点：叶子数 = 左子树叶子数 + 右子树叶子数
    return LeafCount(T->lchild) + LeafCount(T->rchild);
}
```

- **时间复杂度**：O(n)
- **稳定性**：—

## 5. 查找（Search）

### 5.1 顺序查找

```c
/**
 * 顺序查找（线性查找）
 * 在数组 arr[0..n-1] 中查找值为 key 的元素
 * 返回元素下标，未找到返回 -1
 */
int SeqSearch(int arr[], int n, int key) {
    int i;

    for (i = 0; i < n; i++) {
        if (arr[i] == key) {
            return i;                // 找到，返回下标
        }
    }
    return -1;                       // 未找到
}
```

- **时间复杂度**：O(n)；平均查找长度 ASL = (n+1)/2
- **稳定性**：稳定

---

### 5.2 折半查找（二分查找）

```c
/**
 * 折半查找（二分查找）
 * 要求数组 arr[0..n-1] 已按升序排列
 * 返回元素下标，未找到返回 -1
 */
int BinarySearch(int arr[], int n, int key) {
    int low = 0;                     // 左边界
    int high = n - 1;                // 右边界
    int mid;

    while (low <= high) {
        mid = (low + high) / 2;      // 取中间位置

        if (key == arr[mid]) {
            return mid;              // 找到，返回下标
        } else if (key < arr[mid]) {
            high = mid - 1;          // 在左半部分继续查找
        } else {
            low = mid + 1;           // 在右半部分继续查找
        }
    }
    return -1;                       // 未找到
}
```

- **时间复杂度**：O(log₂ n)；平均查找长度 ASL ≈ log₂(n+1) - 1
- **稳定性**：稳定

## 6. 排序（Sort）

### 6.1 冒泡排序（Bubble Sort）

```c
/**
 * 冒泡排序：升序排序
 * 每趟将相邻元素两两比较，将最大的元素"冒泡"到末尾
 */
void BubbleSort(int arr[], int n) {
    int i, j;
    int temp;
    int flag;                        // 标志位，优化：本趟是否发生过交换

    for (i = 0; i < n - 1; i++) {    // 总共 n-1 趟
        flag = 0;                    // 每趟开始时重置标志

        for (j = 0; j < n - 1 - i; j++) {   // 每趟比较次数递减
            if (arr[j] > arr[j + 1]) {      // 相邻元素比较，逆序则交换
                temp = arr[j];
                arr[j] = arr[j + 1];
                arr[j + 1] = temp;
                flag = 1;            // 发生了交换
            }
        }

        if (flag == 0) {             // 本趟没有交换，说明已有序
            break;                   // 提前结束
        }
    }
}
```

- **时间复杂度**：最好 O(n)（已有序），最坏 O(n²)（逆序），平均 O(n²)
- **稳定性**：稳定 ✅

---

### 6.2 选择排序（Selection Sort）

```c
/**
 * 简单选择排序：升序排序
 * 每趟从未排序部分选出最小的元素，放到已排序部分的末尾
 */
void SelectSort(int arr[], int n) {
    int i, j;
    int minIndex;                    // 最小元素的下标
    int temp;

    for (i = 0; i < n - 1; i++) {    // 总共 n-1 趟
        minIndex = i;                // 假设当前 i 为最小元素

        for (j = i + 1; j < n; j++) {       // 在 i 之后查找更小的元素
            if (arr[j] < arr[minIndex]) {
                minIndex = j;        // 更新最小元素下标
            }
        }

        // 将最小元素与第 i 个元素交换
        if (minIndex != i) {
            temp = arr[i];
            arr[i] = arr[minIndex];
            arr[minIndex] = temp;
        }
    }
}
```

- **时间复杂度**：始终 O(n²)（最好、最坏、平均均为 O(n²)）
- **稳定性**：不稳定 ❌（例如 [5, 5, 2] 中第一个 5 会被换到后面）

---

### 6.3 插入排序（Insertion Sort）

```c
/**
 * 直接插入排序：升序排序
 * 将待排序元素插入到已排序序列中的正确位置
 */
void InsertSort(int arr[], int n) {
    int i, j;
    int temp;                        // 暂存待插入元素

    for (i = 1; i < n; i++) {        // 从第 2 个元素开始（下标 1）
        temp = arr[i];               // 取出当前待插入元素
        j = i - 1;                   // 从已排序部分的最后一个开始比较

        // 将大于 temp 的元素依次后移
        while (j >= 0 && arr[j] > temp) {
            arr[j + 1] = arr[j];
            j--;
        }

        arr[j + 1] = temp;           // 将 temp 插入到正确位置
    }
}
```

- **时间复杂度**：最好 O(n)（已有序），最坏 O(n²)（逆序），平均 O(n²)
- **稳定性**：稳定 ✅

### 6.4 快速排序（Quick Sort）

```c
/**
 * 快速排序的核心：划分操作
 * 以第一个元素为基准（pivot），将数组分为左右两部分
 * 返回基准元素最终位置
 */
int Partition(int arr[], int low, int high) {
    int pivot = arr[low];            // 选择第一个元素作为基准

    while (low < high) {
        // 从右向左找第一个小于 pivot 的元素
        while (low < high && arr[high] >= pivot) {
            high--;
        }
        arr[low] = arr[high];        // 将小于 pivot 的元素移到左边

        // 从左向右找第一个大于 pivot 的元素
        while (low < high && arr[low] <= pivot) {
            low++;
        }
        arr[high] = arr[low];        // 将大于 pivot 的元素移到右边
    }

    arr[low] = pivot;                // 基准元素归位
    return low;                      // 返回基准的最终位置
}

/**
 * 快速排序：递归实现
 * 对数组 arr[low..high] 进行升序排序
 */
void QuickSort(int arr[], int low, int high) {
    int pivotPos;                    // 基准位置

    if (low < high) {
        pivotPos = Partition(arr, low, high);   // 划分
        QuickSort(arr, low, pivotPos - 1);      // 递归排序左半部分
        QuickSort(arr, pivotPos + 1, high);     // 递归排序右半部分
    }
}

// 调用入口：QuickSort(arr, 0, n-1);
```

- **时间复杂度**：最好 O(n log₂ n)（每次划分均匀），最坏 O(n²)（有序或逆序），平均 O(n log₂ n)
- **空间复杂度**：O(log₂ n)（递归栈深度）
- **稳定性**：不稳定 ❌（基准元素与 far 元素交换时可能改变相对顺序）

## 考试答题技巧

1. **手写代码规范**：变量名清晰（推荐英文或拼音），缩进整齐，花括号对齐。
2. **注释加分**：关键步骤（如"交换"、"递归"、"查找"）加一行中文注释。
3. **边界检查**：插入/删除操作务必检查位置合法性；栈和队列要检查空/满。
4. **链表操作**：涉及指针操作时，先画图再写代码，避免指针丢失。
5. **排序排序**：默写排序模板时，注意双层循环的边界条件。
6. **递归函数**：先写终止条件（if），再写递归体。
7. **时间不够**：先写出核心框架（函数头、循环结构、关键逻辑），再补充细节。
